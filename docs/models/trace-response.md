# TraceResponse

One agent invocation — what it ran, how it ended, and when.

## Example Usage

```typescript
import { TraceResponse } from "@albus-ts/sdk/models";

let value: TraceResponse = {
  invocationKey: "<value>",
  sessionId: "<id>",
  status: "FAILED",
  spansExpired: true,
  startedAt: new Date("2026-10-15T12:52:59.777Z"),
  failure: {
    kind: "crash",
    message: "The invocation failed. Try again.",
  },
  sessionPosition: 566692,
  spans: [
    {
      id: "<id>",
      type: "step",
      status: "SUCCEEDED",
      startedAt: new Date("2025-01-12T11:57:40.542Z"),
      endedAt: new Date("2026-02-21T12:54:07.863Z"),
    },
  ],
};
```
